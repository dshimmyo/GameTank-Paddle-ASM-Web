Player 1 Controls:

Gamepad:
  • Arrow Keys = Directional Buttons
  • Enter = Start
  • Z, X, C = A, B, C
  • B, N, M = also A, B, C
Paddle (Mouse):
  • Mouse left/right = Paddle left/right

Player 2 Controls:

  • T,F,G,H = Up, Left, Down, Right
  • 1 = Start
  • Tab, Q, W = A, B, C
  • LShift, A, S = also A, B, C

General Keybinds:

  • R = Reset
  • Shift+R = Hard Reset

Assembly Source Code


; --- GameTank Assembly Code ---
DMA_Flags = $2007 ;need to define DMA_Flags keyword
Bank_Flags = $2005 ;controls banking options

GamePad1 = $2008
GamePad2 = $2009

DMA_VX = $4000
DMA_VY = $4001
DMA_WIDTH = $4004
DMA_HEIGHT = $4005
DMA_Status = $4006
DMA_Color = $4007
Framebuffer = $4000

inflate_zp = $F0 ;F1 F2 F3 are reserved addresses for pre-built inflate

Player_X = $0 ;pos
ButtonsPressed0 = $2 ;first read
ButtonsPressed1 = $3 ;second read
PaddleByte = $4
temp_xyz = $5
;gamepad inputs
;INPUT_MASK_UP = 	%00001000 ;both reads - hard wired to pin 1 , bypasses mux
;INPUT_MASK_DOWN = 	%00000100 ;both reads - hard wired to pin 2 , bypasses mux
;INPUT_MASK_LEFT = 	%00000010
;INPUT_MASK_RIGHT = 	%00000001

;paddle inputs
PADDLE_INPUT_MASK_UP = 	%00001000 ; first read ; paddle byte 1
PADDLE_INPUT_MASK_DOWN = 	%00000100 ;first read ; paddle byte 2
PADDLE_INPUT_MASK_LEFT = 	%00000010 ; second read ; paddle byte 3
PADDLE_INPUT_MASK_RIGHT = 	%00000001 ;second read ; paddle byte 4
PADDLE_INPUT_MASK_X = 	%00000010 ; first read ; paddle byte 8
PADDLE_INPUT_MASK_Y = 	%00000100 ; first read ; paddle byte 16
PADDLE_INPUT_MASK_Z = 	%00001000 ; first read ; paddle byte 32
PADDLE_INPUT_MASK_MODE = 	%00000001 ; first read ; paddle byte 64

;paddle byte positions for reference
;Think of the paddle byte as a rotation value

;PADDLE_BYTE_UP = %00000001
;PADDLE_BYTE_DOWN = %00000010
;PADDLE_BYTE_LEFT = %00000100
;PADDLE_BYTE_RIGHT = %00001000
;PADDLE_BYTE_X = %00010000
;PADDLE_BYTE_Y = %00100000
;PADDLE_BYTE_Z = %01000000
;PADDLE_BYTE_MODE = %10000000

; Note: the paddle bits are inverted so after the byte is constructed
; you still need to invert it for the value to read 0->255 where low is turned all the way CCW and high is CW

	.org $E000
	;organize: tells the assembler to put code at a specific address
	;influences address values for labels and production of final rom
	;$E000 or 57344 is the starting location for an 8KB cartridge

RESET:
	;code here will get run once when the system boots
	STZ Bank_Flags
	
	CLI

	LDA #$FD
	STA DMA_Flags
	LDA #1
	STA DMA_WIDTH
	STA DMA_HEIGHT
	STZ DMA_VX
	STZ DMA_VY
	STA DMA_Status
	STZ temp_xyz
	WAI

	LDA #%10000000 ;Activate lower page of VRAM/GRAM, CPU accesses GRAM, no IRQ , no transparency
	STA DMA_Flags

	STZ Player_X
	STZ ButtonsPressed0
	STZ ButtonsPressed1
	STZ PaddleByte

	JSR FillBackground
	WAI

	CLI ; clear interrupt - disables all interrupts

;# indicates using a value directly instead of an address
;% denotes binary notation. Each bit is a switch
;right to left:
;1 *DMA_enabled - dma blitter engine
;2 video out page
;4 *NMI Enabled - trigger an NMI event every vsync
;8 *color fill mode enable
;16 tile mode disable
;32 CPU bank select
;64 *enable copy completion IRQ - trigger IRQ event when a copy operation completes
;128 transparency copy disable (skips zeroes)

Forever:
	;code here will run over and over until the system is reset or turned off
	LDA GamePad2 ;resets gamepad1 selet signal 

	;first read
	LDA GamePad1 ;first read select Low up,down,a,start
	EOR #$FF ; exclusive or inverts each bit - remember this because paddle bits are inverted
	STA ButtonsPressed0

	;second read
	LDA GamePad1 ;second read select High up,down,left,right,b,c
	EOR #$FF ; exclusive or inverts each bit - remember this because paddle bits are inverted
	STA ButtonsPressed1
	
	;paddle handling
	;compile a paddle byte that looks like mode,z,y,x,rt,lf,dn,up
	STZ PaddleByte ; init paddlebyte
	LDA ButtonsPressed0 ; up is in the first read
	AND #PADDLE_INPUT_MASK_UP ;4th bit
	LSR
	LSR
	LSR ;shift down to bit 1
	STA PaddleByte ; 

	LDA ButtonsPressed0 ; down - first read
	AND #PADDLE_INPUT_MASK_DOWN ; 3rd bit
	LSR ;bit 2
	ORA PaddleByte
    STA PaddleByte

	LDA ButtonsPressed1 ; left in second read
	AND #PADDLE_INPUT_MASK_LEFT ;2nd bit
	ASL ; bit 3
	ORA PaddleByte
    STA PaddleByte

	LDA ButtonsPressed1 ; right in second read
	AND #PADDLE_INPUT_MASK_RIGHT ; bit 1
	ASL
	ASL
	ASL ;bit 4
	ORA PaddleByte
    STA PaddleByte

	; combining xyz for faster bit shifting
	LDA ButtonsPressed0 ; xyzmode are all in the first read
	AND #PADDLE_INPUT_MASK_X ;bit2
	STA temp_xyz 

	LDA ButtonsPressed0 ; xyzmode are all in the first read
	AND #PADDLE_INPUT_MASK_Y ;bit3
	ORA temp_xyz
	STA temp_xyz

	LDA ButtonsPressed0 ; xyzmode are all in the first read
	AND #PADDLE_INPUT_MASK_Z ;bit4
	ORA temp_xyz

	;shift xyz as a group
	ASL
	ASL
	ASL ; bit 7
	; add to the pack
	ORA PaddleByte
    STA PaddleByte

	LDA ButtonsPressed0 ; xyzmode are all in the first read
	AND #PADDLE_INPUT_MASK_MODE ; bit1
	ASL ; shift left by 1
	ASL 
	ASL
	ASL
	ASL
	ASL
	ASL ; bit 8
	ORA PaddleByte
    STA PaddleByte

	EOR #$FF ; invert paddlebyte - If you are paying attention it is inverted twice
	STA PaddleByte ; keeps the 8-bit paddle rotation value if needed
	
	LSR ;shift right by 1 so we only get 128 rotation values, lines up with horizontal resolution
	STA Player_X

;AfterInputs:
	JSR FillBackground
	WAI
	JSR DrawPlayer
	WAI ; wait for the irq after the draw event (64 cycles for 64 pixels square)
	; wait for nmi at the start of vsync (60 fps) start of the frame being sent to the tv
	WAI
	JMP Forever

	; subroutines

FillBackground:
	LDA #%01001101 ;load value into A register aka accumulator ; disable color fill bit 4 from right
	STA DMA_Flags ;store value from A into DMA_Flags
	STZ DMA_VX ; stores zero?
	STZ DMA_VY

	LDA #127 ; #8
	STA DMA_WIDTH ;same value of 8 stored in width and height to save time
	STA DMA_HEIGHT

	LDA #$FE ;#%11111110 ; clyde chose #$FE
	STA DMA_Color

	LDA #1 ;initate draw operation
	STA DMA_Status ; turn off the event signal. needs to be cleared after the draw otherwise it gets stuck in the irq event

	RTS

DrawPlayer:
	LDA #%01001101 ;load value into A register aka accumulator, 4th bit color fill mode enabled
	STA DMA_Flags ;store value from A into DMA_Flags
	LDA Player_X ;fetch the byte at address 0
	
	STA DMA_VX
	LDA #24
	STA DMA_VY
	
	LDA #16
	STA DMA_WIDTH ;same value of 8 stored in width and height to save time
	STA DMA_HEIGHT

	LDA #$23
	STA DMA_Color

	LDA #1 ;initate draw operation
	STA DMA_Status ; turn off the event signal. needs to be cleared after the draw otherwise it gets stuck in the irq event
	RTS

IRQ:
	;code here will run after certain interrupts (if enabled)
	;such as when a DMA draw call completes

	STZ DMA_Status ; turn off the event signal
	RTI
NMI:
	;code here will run once per frame (if enabled)
	RTI ;return immediately

	.org $FFFA
	.dw NMI ;non-maskable interrupt
	.dw RESET ;starting point for the program
	.dw IRQ ;interrupt request
	;.dw "dataword" can be given a number or label, places raw data in the binary output
